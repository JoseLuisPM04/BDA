# ------------------------------------------------------------------------------
# PASO 3: CONSULTAS DE DATOS (READ OPERATIONS)
# ------------------------------------------------------------------------------

# 1. Consulta puntual por RowKey (Obtener toda la información del estudiante EST_005)
get 'estudiantes_bigdata', 'EST_005'

# 2. Consulta puntual de columnas específicas (Solo nombre y nota del estudiante EST_010)
get 'estudiantes_bigdata', 'EST_010', {COLUMNS => ['personal:nombre', 'academico:nota']}

# 3. Conteo total de filas en la tabla
count 'estudiantes_bigdata'

# 4. Escaneo acotado por rango de RowKeys (Obtener estudiantes del EST_001 al EST_010)
# Nota: STOPROW es exclusivo, por lo que usamos 'EST_011' para incluir 'EST_010'.
scan 'estudiantes_bigdata', {STARTROW => 'EST_001', STOPROW => 'EST_011'}

# 5. Escaneo proyectando solo una Familia de Columnas (Ver solo datos académicos de los primeros 5 alumnos)
scan 'estudiantes_bigdata', {STARTROW => 'EST_001', STOPROW => 'EST_006', COLUMNS => 'academico'}

# 6. Escaneo con filtro de valor (Buscar estudiantes cuya ciudad sea 'Valencia')
scan 'estudiantes_bigdata', {FILTER => "SingleColumnValueFilter('personal', 'ciudad', =, 'binary:Valencia')"}


# ------------------------------------------------------------------------------
# PASO 4: ACTUALIZACIONES DE DATOS Y CONTROL DE VERSIONES (UPDATES)
# ------------------------------------------------------------------------------
# En HBase, NO existe el comando 'UPDATE'. Para actualizar un dato se realiza
# un nuevo 'put' sobre la misma RowKey y columna. HBase añade la nueva versión
# con un timestamp más reciente sin borrar de inmediato la anterior.

# Actualización 1: Subir la nota del estudiante EST_005 de su valor original a 9.5
put 'estudiantes_bigdata', 'EST_005', 'academico:nota', '9.5'

# Actualización 2: Volver a corregir la nota del estudiante EST_005 a 10.0 (Tercera versión)
put 'estudiantes_bigdata', 'EST_005', 'academico:nota', '10.0'

# Ver la versión más reciente del estudiante EST_005
get 'estudiantes_bigdata', 'EST_005', {COLUMNS => 'academico:nota'}

# Consultar EL HISTORIAL DE VERSIONES (gracias a VERSIONS => 3 configurado en el 'create')
get 'estudiantes_bigdata', 'EST_005', {COLUMNS => 'academico:nota', VERSIONS => 3}


# ------------------------------------------------------------------------------
# PASO 5: ELIMINACIÓN PARCIAL Y TOTAL DE CELDAS (DELETE)
# ------------------------------------------------------------------------------

# 1. Borrado de un campo/columna específica (Eliminar el teléfono del estudiante EST_020)
delete 'estudiantes_bigdata', 'EST_020', 'contacto:telefono'

# Verificar que el teléfono ha sido marcado con una lápida (Tombstone) y ya no se muestra
get 'estudiantes_bigdata', 'EST_020'

# 2. Borrado completo de una fila entera (Eliminar al estudiante EST_200)
deleteall 'estudiantes_bigdata', 'EST_200'

# Verificar que el registro EST_200 ya no existe
get 'estudiantes_bigdata', 'EST_200'


# ------------------------------------------------------------------------------
# PASO 6: PERSISTENCIA Y MANTENIMIENTO EN DISCO
# ------------------------------------------------------------------------------
